// Edge Function: create-checkout // Recebe o total do carrinho do frontend, cria uma "payment order" na Viva // (VivaWallet) e devolve o URL para onde redirecionar o cliente. // // Segredos necessários (definir em Supabase → Project Settings → Edge Functions → Secrets): // VIVA_CLIENT_ID — Client ID do Smart Checkout // VIVA_CLIENT_SECRET — Client Secret do Smart Checkout // VIVA_SOURCE_CODE — código da fonte de pagamento (Vendas → Fontes de Pagamento) // VIVA_ENV — "demo" ou "production" (usa "demo" até testares tudo) // ALLOWED_ORIGIN — o domínio da tua loja (ex: https://SEUUSER.github.io) import { serve } from "https://deno.land/std@0.224.0/http/server.ts"; function corsHeaders(origin: string){ return { "Access-Control-Allow-Origin": origin, "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; } async function fetchWithTimeout(url: string, init: RequestInit, ms = 10000){ const ctrl = new AbortController(); const t = setTimeout(() => ctrl.abort(), ms); try { return await fetch(url, { ...init, signal: ctrl.signal }); } finally { clearTimeout(t); } } async function fetchWithRetry(url: string, init: RequestInit, attempts = 3){ let lastErr: unknown; for (let i = 0; i < attempts; i++){ try { return await fetchWithTimeout(url, init); } catch (e){ lastErr = e; if (i < attempts - 1) await new Promise(r => setTimeout(r, 400 * (i + 1))); } } throw lastErr; } serve(async (req) => { const allowedOrigin = Deno.env.get("ALLOWED_ORIGIN") || "*"; const headers = corsHeaders(allowedOrigin); if (req.method === "OPTIONS") return new Response("ok", { headers }); try { const { amountEur, customerName, customerEmail, description } = await req.json(); if (!amountEur || amountEur <= 0){ return new Response(JSON.stringify({ error: "amountEur inválido" }), { status: 400, headers }); } const env = (Deno.env.get("VIVA_ENV") || "demo").toLowerCase(); const accountsHost = env === "production" ? "accounts.vivapayments.com" : "demo-accounts.vivapayments.com"; const apiHost = env === "production" ? "www.vivapayments.com" : "demo.vivapayments.com"; const clientId = (Deno.env.get("VIVA_CLIENT_ID") || "").trim(); const clientSecret = (Deno.env.get("VIVA_CLIENT_SECRET") || "").trim(); const sourceCode = (Deno.env.get("VIVA_SOURCE_CODE") || "").trim(); // 1. Obter access token OAuth2 (client_credentials) const basic = btoa(`${clientId}:${clientSecret}`); let tokenRes: Response; try { tokenRes = await fetchWithRetry(`https://${accountsHost}/connect/token`, { method: "POST", headers: { "Authorization": `Basic ${basic}`, "Content-Type": "application/x-www-form-urlencoded", "Accept": "application/json", "User-Agent": "DonTaskoStore/1.0 (+Supabase Edge Function)", }, body: "grant_type=client_credentials", }); } catch (fetchErr){ return new Response(JSON.stringify({ error: "Falha de rede ao obter token OAuth2", detail: String(fetchErr) }), { status: 502, headers }); } if (!tokenRes.ok){ const t = await tokenRes.text(); return new Response(JSON.stringify({ error: "Falha a obter token OAuth2", detail: t }), { status: 502, headers }); } const { access_token } = await tokenRes.json(); // 2. Criar a payment order // Corpo minimizado: só inclui email/nome do cliente se existirem // (evita enviar chaves com valor undefined) e cabeçalhos extra // (User-Agent, Accept, Accept-Encoding: identity) para testar se o // erro de rede é causado por um WAF/CDN da Viva a bloquear o // "fingerprint" por omissão do cliente fetch do Deno. const customerObj: Record = { requestLang: "pt-PT" }; if (customerEmail) customerObj.email = customerEmail; if (customerName) customerObj.fullName = customerName; const orderBody = JSON.stringify({ amount: Math.round(amountEur * 100), // cêntimos customerTrns: description || "Encomenda na loja", customer: customerObj, sourceCode, }); let orderRes: Response; try { orderRes = await fetchWithRetry(`https://${apiHost}/checkout/v2/orders`, { method: "POST", headers: { "Authorization": `Bearer ${access_token}`, "Content-Type": "application/json", "Accept": "application/json", "Accept-Encoding": "identity", "User-Agent": "DonTaskoStore/1.0 (+Supabase Edge Function)", }, body: orderBody, }, 3); } catch (fetchErr){ return new Response(JSON.stringify({ error: "Falha de rede ao criar a order (mesmo depois de 3 tentativas)", detail: String(fetchErr), debug_api_host: apiHost, debug_order_body: orderBody, }), { status: 502, headers }); } if (!orderRes.ok){ const t = await orderRes.text(); return new Response(JSON.stringify({ error: "Falha a criar a order", detail: t || "(resposta vazia da Viva)", debug_status: orderRes.status, debug_status_text: orderRes.statusText, debug_headers: Object.fromEntries(orderRes.headers.entries()), debug_order_body: orderBody, }), { status: 502, headers }); } const { orderCode } = await orderRes.json(); const checkoutUrl = `https://${apiHost}/web/checkout?ref=${orderCode}`; return new Response(JSON.stringify({ checkoutUrl, orderCode }), { headers: { ...headers, "Content-Type": "application/json" }, }); } catch (e){ return new Response(JSON.stringify({ error: String(e) }), { status: 500, headers }); } });
